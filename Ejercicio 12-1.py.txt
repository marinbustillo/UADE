# Importa el dataset Iris incluido en Scikit-learn.
from sklearn.datasets import load_iris
# Importa la función que divide datos en entrenamiento y prueba.
from sklearn.model_selection import train_test_split
# Importa el escalador que estandariza las características.
from sklearn.preprocessing import StandardScaler
# Importa el clasificador de K vecinos más cercanos.
from sklearn.neighbors import KNeighborsClassifier
# Importa Pipeline para encadenar escalado y modelo.
from sklearn.pipeline import Pipeline
# Importa las métricas de clasificación.
from sklearn.metrics import accuracy_score, confusion_matrix
# Carga las características en X y las clases en y.
X, y = load_iris(return_X_y=True)
# Separa 80% para entrenamiento y 20% para prueba.
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=42, stratify=y
)
# Crea un Pipeline que primero escala y después clasifica.
modelo = Pipeline([
    ("escalador", StandardScaler()),
    ("knn", KNeighborsClassifier(n_neighbors=3))
])
# Ajusta el escalador y el clasificador usando train.
modelo.fit(X_train, y_train)
# Predice las clases de las observaciones de prueba.
y_pred = modelo.predict(X_test)
# Calcula la proporción de predicciones correctas.
accuracy = accuracy_score(y_test, y_pred)
# Construye la matriz de confusión.
matriz = confusion_matrix(y_test, y_pred)
# Muestra el accuracy redondeado a tres decimales.
print("Accuracy:", round(accuracy, 3))
# Muestra la matriz de confusión completa.
print("Matriz de confusión:\n", matriz)
# Muestra las primeras cinco clases reales.
print("Reales:", y_test[:5])
# Muestra las primeras cinco clases predichas.
print("Predichas:", y_pred[:5])
