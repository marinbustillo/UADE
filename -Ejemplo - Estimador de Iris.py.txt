# Importa la función que carga el dataset Iris integrado.
from sklearn.datasets import load_iris

# Carga Iris y solicita estructuras de pandas.
iris = load_iris(as_frame=True)

# Guarda las características predictoras en X.
X = iris.data

# Guarda la especie de cada flor en y.
y = iris.target

# Muestra las dimensiones de X: filas y columnas.
print("Dimensiones de X:", X.shape)

# Muestra los nombres de las variables predictoras.
print("Características:", iris.feature_names)

# Muestra las primeras cinco observaciones.
print(X.head())

# Muestra cuántos ejemplos existen de cada clase.
print(y.value_counts().sort_index())

# Importa la función para separar entrenamiento y prueba.
from sklearn.model_selection import train_test_split

# Importa el escalador que estandariza cada característica.
from sklearn.preprocessing import StandardScaler

# Divide X e y; 20% queda reservado para prueba.
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=42, stratify=y
)

# Crea el objeto que aprenderá media y desviación estándar.
scaler = StandardScaler()

# Aprende el escalado con entrenamiento y transforma X_train.
X_train_scaled = scaler.fit_transform(X_train)

# Usa los mismos parámetros para transformar X_test.
X_test_scaled = scaler.transform(X_test)

# Muestra cuántos casos quedaron en cada conjunto.
print("Train:", X_train.shape, "Test:", X_test.shape)

# Importa el clasificador de regresión logística.
from sklearn.linear_model import LogisticRegression

# Importa métricas para evaluar las predicciones.
from sklearn.metrics import accuracy_score, confusion_matrix

# Crea el modelo y aumenta el máximo de iteraciones.
model = LogisticRegression(max_iter=300)

# Entrena el modelo con los datos escalados de entrenamiento.
model.fit(X_train_scaled, y_train)

# Genera predicciones para observaciones no vistas.
y_pred = model.predict(X_test_scaled)

# Calcula la proporción de predicciones correctas.
accuracy = accuracy_score(y_test, y_pred)

# Construye la matriz de confusión real frente a predicho.
matrix = confusion_matrix(y_test, y_pred)

# Muestra las métricas obtenidas.
print("Accuracy:", round(accuracy, 3))
print("Matriz de confusión:\n", matrix)
